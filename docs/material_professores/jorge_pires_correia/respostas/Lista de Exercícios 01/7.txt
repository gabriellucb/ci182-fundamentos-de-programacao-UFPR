'''
7. Construa um programa que, tendo como dados de entrada dois pontos
quaisquer no plano, P(x1 ,y1 ) e P(x 2 ,y2 ), escreva a distância entre eles. A
fórmula que efetua tal cálculo é: (consultar slide)
'''

import math

x1 = int (input ("Digite a coordenada X do primeiro ponto: "))
y1 = int (input ("Digite a coordenada Y do primenro ponto: "))
x2 = int (input ("Digite a coordenada X do segundo ponto: "))
y2 = int (input ("Digite a coordenada Y do segundo ponto: "))

distancia = math.sqrt ( ((x2 - x1) ** 2) + ((y2 - y1) ** 2) )

print ("A distância entre os pontos é ", distancia)
