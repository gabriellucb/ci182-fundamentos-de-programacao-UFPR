'''
Fazer um programa em Python para calcular a soma dos cinquenta primeiros números ímpares. Ao final, imprimir esta soma.
'''

soma_total = 0
numero_impar_atual = 1
for i in range (50):
	soma_total = soma_total + numero_impar_atual
	numero_impar_atual = numero_impar_atual + 2

print ("Soma dos 50 primeiros numeros impares:", soma_total)
