'''
Fazer um programa em Python para calcular e imprimir todos os números primos entre 1 e 100.
'''


soma_total = 0
for i in range (1, 101):
	# precisamos verificar se o numero atual eh primo. Para isso, vamos iterar de 2 ate o numero atual para realizar essa verificacao.
	if i <=1:
		eh_primo = False
	else:
		eh_primo = True
		for j in range (2, i):
			if i % j  == 0:
				eh_primo = False
				break

	# se o numero for maior que 1 e passou por todas as iteracoes sem entrar no if, quer dizer que o numero eh primo e que a variavel eh_primo esta com valor True
	if eh_primo == True:
		soma_total = soma_total + i
		
print ("Soma de todos os numeros primeros entre 1 e 100:", soma_total)
